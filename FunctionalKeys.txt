@Keyword
def spinnerWaitToDisappear(
        int timeToWait = 30,
        int noLoadWait = 3,
        boolean withServerLoad = true,
        int retryCount = 2
) {

    TestObject spinner = findTestObject('Object Repository/X360/Misc/Performance/Generic/spinner_generic')
    int maxWait = withServerLoad ? timeToWait : noLoadWait
    int attempt = 0

    while (attempt <= retryCount) {

        WebUI.comment("Spinner wait attempt ${attempt + 1} of ${retryCount + 1}")

        // If spinner not present → proceed immediately
        if (!WebUI.verifyElementPresent(spinner, 1, FailureHandling.OPTIONAL)) {
            WebUI.comment("Spinner not found in DOM. Proceeding immediately.")
            return true
        }

        WebUI.comment("Spinner detected. Polling for disappearance (max ${maxWait}s)...")

        long start = System.currentTimeMillis()

        while ((System.currentTimeMillis() - start) / 1000 < maxWait) {

            boolean present = WebUI.verifyElementPresent(spinner, 0, FailureHandling.OPTIONAL)
            boolean visible = WebUI.verifyElementVisible(spinner, FailureHandling.OPTIONAL)

            if (!present || !visible) {
                long elapsed = (System.currentTimeMillis() - start) / 1000
                WebUI.comment("Spinner disappeared after ~${elapsed}s.")
                return true
            }

            // Small polling interval (fast response, low CPU)
            WebUI.delay(0.2)
        }

        WebUI.comment("Spinner did NOT disappear within ${maxWait}s. Retrying...")
        attempt++
        WebUI.delay(1)
    }

    WebUI.comment("Spinner did NOT disappear after all ${retryCount + 1} attempts.")
    return false
}

		}

			KeywordUtil.logInfo("Toaster did not disappear within ${timeout} seconds.")
			WebUI.click(closeBtn, FailureHandling.OPTIONAL)
			return false
		}

		KeywordUtil.logInfo("Toaster appeared and disappeared successfully.")
		return true
	}

@Keyword
def pinCollapsiblePanel() {

    spinnerWaitToDisappear(GlobalVariable.sTimeToWait, GlobalVariable.iLoad, GlobalVariable.withServerLoad)
    WebUI.waitForPageLoad(GlobalVariable.iLoad)

    def panelParams = [
        'sClass1': 'side-filter main-side-filter',
        'sClass2': 'collapsible-side-panel'
    ]

    def pinParams = [
        'sValue1': 'side-filter main-side-filter',
        'sID'    : 'sideFilterPin'
    ]

    TestObject panel = findTestObject('X360/Misc/Performance/Generic/pin_generic', panelParams)
    TestObject pin   = findTestObject('X360/Misc/Performance/Generic/pinParam_generic', pinParams)

    // Check if pin is already visible BEFORE doing anything else
    boolean pinInitiallyVisible = WebUI.verifyElementVisible(pin, FailureHandling.OPTIONAL)

    if (pinInitiallyVisible) {
        WebUI.comment("Pin already visible — skipping panel expansion and retry logic.")

        if (WebUI.verifyElementClickable(pin, FailureHandling.OPTIONAL)) {
            WebUI.comment("Pinned the collapsible panel.")
            return true
        } else {
            KeywordUtil.markFailed("Pin is visible but not clickable.")
            return false
        }
    }

    // STEP 1 — Expand panel (only if pin was not visible)
    if (WebUI.verifyElementClickable(panel, FailureHandling.OPTIONAL)) {
        WebUI.click(panel)
        WebUI.comment("Expanded collapsible panel.")
    } else {
        WebUI.comment("Panel is not clickable.")
        KeywordUtil.markFailed("Panel could not be expanded.")
    }

    // STEP 2 — Retry logic (only reached if pin was NOT initially visible)
    int retries = 5
    boolean clicked = false

    for (int i = 0; i < retries; i++) {

        boolean pinVisible   = WebUI.waitForElementVisible(pin, 2, FailureHandling.OPTIONAL)
        boolean pinClickable = WebUI.waitForElementClickable(pin, 2, FailureHandling.OPTIONAL)

        if (pinVisible && pinClickable) {
            WebUI.click(pin)
            WebUI.comment("Pinned the collapsible panel.")
            clicked = true
            break
        }

        spinnerWaitToDisappear(GlobalVariable.sTimeToWait, GlobalVariable.iLoad, GlobalVariable.withServerLoad)
    }

    if (!clicked) {
        KeywordUtil.markFailed("Pin icon never became clickable.")
    }

    return clicked
}

		}
	}

	@Keyword
	def generateRandomString(Map args = [:]) {
		String alphanum_characters = args.get('alphanum_characters') ?: "AB1CDE2FGHIJ7KLMN3OPQR4STUV895WX60YZ"
		int stringSize = args.get('stringSize') ?: 5
		String prefix = args.get('prefix') ?: "auto_"

		def randomString = (1..stringSize).collect {
			def randomIndex = (int) (Math.random() * alphanum_characters.length())
			alphanum_characters[randomIndex]
		}.join()

		return prefix + randomString
	}
