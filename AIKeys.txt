@Keyword
def openAiAssist(boolean perfEnabled) {

    def busy = findTestObject(
        'Object Repository/X360/XAI/Performance/AiAssist/03_General/div_topSectionWithChatBusy'
    )

    TestObject btnAi =
        findTestObject('X360/XAI/Performance/AiAssist/03_General/btn_AiAssist')

    WebUI.waitForElementClickable(btnAi, GlobalVariable.iLoad)

    /* -------------------------------
       Install JS network hook (once)
       ------------------------------- */
    if (perfEnabled) {
		WebUI.executeJavaScript("""
(function () {
    if (window.__aiTimingInstalled) return;

    window.__aiTimingInstalled = true;
    window.__aiApiCallTime = null;

    const originalFetch = window.fetch;
    window.fetch = function () {
        const url = arguments[0];
        if (typeof url === 'string' &&
            url.includes('/api-ui/ai/')) {
            window.__aiApiCallTime ??= Date.now();
        }
        return originalFetch.apply(this, arguments);
    };

    const originalOpen = XMLHttpRequest.prototype.open;
    XMLHttpRequest.prototype.open = function (method, url) {
        if (typeof url === 'string' &&
            url.includes('/api-ui/ai/')) {
            window.__aiApiCallTime ??= Date.now();
        }
        return originalOpen.apply(this, arguments);
    };
})();
""", null)
    }

    /* ✅ Reset timing BEFORE click */
    if (perfEnabled) {
        WebUI.executeJavaScript(
            "window.__aiStreamPromptTime = null;", null
        )
    }

    /* -------------------------------
       Click AI Assist
       ------------------------------- */
    long clickTime = System.currentTimeMillis()
    WebUI.click(btnAi)

    /* -------------------------------
       Measure click → AI API call
       ------------------------------- */
    if (perfEnabled) {

        long duration = -1
        long timeoutMs = GlobalVariable.iLoad * 1000
        long waitStart = System.currentTimeMillis()

        while (true) {

            /* ✅ Hard timeout guard */
            if ((System.currentTimeMillis() - waitStart) >= timeoutMs) {
                KeywordUtil.logInfo(
                    'AI streamPrompt not detected within timeout'
                )
                break
            }


				def apiTs = WebUI.executeJavaScript(
				    "return window.__aiApiCallTime;", null
				)
				
				if (apiTs instanceof Number) {
				    duration = apiTs.longValue() - clickTime
				}


            /* ✅ Prevent log flooding / CPU spin */
            Thread.sleep(200)
        }

        if (duration > 0) {

            GlobalVariable.CellValue =
                'AI Assist Startup Time'

            def df = new java.text.SimpleDateFormat("HH:mm:ss.SSS")
            GlobalVariable.startTime =
                df.format(new Date(clickTime))
            GlobalVariable.endTime =
                df.format(new Date(clickTime + duration))

		CustomKeywords.'x360.Performance_ExcelFunctions.writeToExcel'()
			
        } else {
            KeywordUtil.logInfo(
                'AI streamPrompt timing not captured'
            )
        }
    }

    /* -------------------------------
       Continue existing AI logic
       ------------------------------- */
    WebUI.switchToDefaultContent()

    WebUI.switchToFrame(
        findTestObject('X360/XAI/Performance/AiAssist/03_General/iframe_Xai'),
        GlobalVariable.iLoad
    )

    perfStart('Open AI - Initial Response Received', perfEnabled)
    CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()

    WebUI.waitForElementPresent(
        findTestObject('X360/XAI/Performance/AiAssist/03_General/div_UseThisResponse'),
        GlobalVariable.sTimeToWait
    )

    perfEnd(perfEnabled)

    WebUI.waitForElementNotPresent(
        busy,
        GlobalVariable.sTimeToWait
    )

    verifyAIResponse(
        findTestObject('X360/XAI/Performance/AiAssist/03_General/div_AiResponseError'),
        findTestObject('X360/XAI/Performance/AiAssist/03_General/div_AiResponseText')
    )
}
	
def verifyAIResponse(TestObject errorResponse, TestObject aiResponse) {

    // Known application/system error phrases only
    List<String> knownErrors = [
        "something went wrong",
        "unexpected error",
        "service unavailable",
        "failed to generate",
        "unable to generate",
        "try again later",
        "request failed",
        "internal server error",
        "timed out",
        "network error"
    ]

    /* --------------------------------------------------
     * First: Validate AI response exists
     * -------------------------------------------------- */

    List<WebElement> responses =
            WebUiCommonHelper.findWebElements(
                aiResponse,
                GlobalVariable.iLoad
            )

    if (!responses.isEmpty()) {

        String responseText = responses.collect {
            it.getAttribute("innerText")?.trim() ?: ""
        }.findAll {
            it
        }.join("\n")

        if (responseText?.trim()) {

            println "================ AI RESPONSE ================="
            println responseText.take(1000)
            println "=============================================="

            WebUI.comment("AI response received:\n${responseText}")

            // Valid response exists
            // Do NOT fail simply because an error object exists somewhere
            return
        }
    }

    /* --------------------------------------------------
     * Second: Check actual visible error messages
     * -------------------------------------------------- */

    List<WebElement> errors =
            WebUiCommonHelper.findWebElements(
                errorResponse,
                FailureHandling.OPTIONAL
            )

    String actualError = null

    for (WebElement err : errors) {

        try {

            if (!err.isDisplayed()) {
                continue
            }

            String text = err.getText()?.trim()

            if (!text) {
                continue
            }

            println "Potential Error Text Found:"
            println text

            boolean isKnownError =
                    knownErrors.any {
                        text.toLowerCase().contains(it)
                    }

            if (isKnownError) {
                actualError = text
                break
            }

        } catch (Exception ignored) {
            println "Skipping stale/invalid error element."
        }
    }

    if (actualError) {

        WebUI.switchToDefaultContent()

        KeywordUtil.markFailedAndStop(
            "AI application error detected:\n${actualError}"
        )
    }

    /* --------------------------------------------------
     * Third: Fallback condition
     * -------------------------------------------------- */

    WebUI.switchToDefaultContent()

    KeywordUtil.markFailedAndStop(
        "No AI response was returned and no known application error was detected."
    )
}

				handleArtifactFlow(Performance_AiAssist_Keywords.AiAssistConstants.LIBRARY_ARTIFACT_NAME)

				startOverConversation()
				break
	
	
			default:
				WebUI.comment("Invalid artifact selection: $userSelection")
		}
	}

def handleArtifactFlow(String artifactName) {

    int TIMEOUT = Math.max(GlobalVariable.iLoad, 10)
    boolean perfEnabled = true

    // Set CellValue based on artifact type
    if (artifactName.equalsIgnoreCase(
        Performance_AiAssist_Keywords.AiAssistConstants.ATTACHED_ARTIFACT_NAME)) {

        perfStart(
            Performance_AiAssist_Keywords.AiAssistConstants.USE_ATTACHED_ARTIFACT,
            perfEnabled
        )

    } else if (artifactName.equalsIgnoreCase(
        Performance_AiAssist_Keywords.AiAssistConstants.LIBRARY_ARTIFACT_NAME)) {

        perfStart(
            Performance_AiAssist_Keywords.AiAssistConstants.SELECT_FROM_LIBRARY,
            perfEnabled
        )
    }

    println "Artifact name received: [${artifactName}]"
    println "Expected library artifact name: [${Performance_AiAssist_Keywords.AiAssistConstants.LIBRARY_ARTIFACT_NAME}]"
    println "TIMEOUT = ${TIMEOUT}"

    TestObject checkbox = findTestObject(
        "Object Repository/X360/XAI/Performance/AiAssist/03_General/checkbox_PageRelatedArtifact",
        [('sValue'): artifactName]
    )

    TestObject insertBtn = findTestObject(
        "Object Repository/X360/Misc/Performance/Generic/button_generic",
        [('sValue'): 'button insertImage']
    )

    WebUI.switchToDefaultContent()

    try {

        boolean checkboxPresent = false

        // Retry because the artifact row may still be rendering
        for (int attempt = 1; attempt <= 3; attempt++) {

            println "Attempt ${attempt} to locate artifact checkbox..."

            checkboxPresent = WebUI.waitForElementPresent(
                checkbox,
                10,
                FailureHandling.OPTIONAL
            )

            if (checkboxPresent) {
                println "Checkbox found on attempt ${attempt}"
                break
            }

            println "Checkbox not found on attempt ${attempt}. Waiting before retry..."
            WebUI.delay(2)
        }

        println "Checkbox present = ${checkboxPresent}"

        if (!checkboxPresent) {

            WebUI.takeScreenshot()

            boolean exists = WebUI.verifyElementPresent(
                checkbox,
                5,
                FailureHandling.OPTIONAL
            )

            boolean visible = WebUI.verifyElementVisible(
                checkbox,
                FailureHandling.OPTIONAL
            )

            println "Checkbox exists after retries: ${exists}"
            println "Checkbox visible after retries: ${visible}"

            KeywordUtil.markFailedAndStop(
                "Artifact checkbox not found after retries. Artifact Name: ${artifactName}"
            )
        }

        WebUI.waitForElementVisible(
            checkbox,
            10,
            FailureHandling.STOP_ON_FAILURE
        )

        WebUI.waitForElementClickable(
            checkbox,
            10,
            FailureHandling.STOP_ON_FAILURE
        )

        println "Clicking artifact checkbox..."
        WebUI.click(checkbox)

        WebUI.waitForElementVisible(
            insertBtn,
            10,
            FailureHandling.STOP_ON_FAILURE
        )

        WebUI.waitForElementClickable(
            insertBtn,
            10,
            FailureHandling.STOP_ON_FAILURE
        )

        println "Clicking Insert button..."
        WebUI.click(insertBtn)

    } catch (Exception e) {

        println "================ ERROR DETAILS ================"
        println "Artifact Name: ${artifactName}"
        println "Exception Type: ${e.getClass().getName()}"
        println "Exception Message: ${e.getMessage()}"
        println "==============================================="

        WebUI.takeScreenshot()

        boolean exists = WebUI.verifyElementPresent(
            checkbox,
            5,
            FailureHandling.OPTIONAL
        )

        boolean visible = WebUI.verifyElementVisible(
            checkbox,
            FailureHandling.OPTIONAL
        )

        println "Checkbox exists after failure: ${exists}"
        println "Checkbox visible after failure: ${visible}"

        KeywordUtil.markFailedAndStop(
            "Failed during artifact selection for '${artifactName}'. Reason: ${e.getMessage()}"
        )
    }

    interactWithAI(
        findTestObject("X360/XAI/Performance/AiAssist/03_General/iframe_Xai"),
        findTestObject("X360/XAI/Performance/AiAssist/03_General/div_UseThisResponse"),
        findTestObject("X360/XAI/Performance/AiAssist/03_General/textarea_AiQuestionbox"),
        findTestObject("X360/XAI/Performance/AiAssist/03_General/img_ButtonSend"),
        findTestObject("X360/XAI/Performance/AiAssist/03_General/div_AiResponseError"),
        findTestObject("X360/XAI/Performance/AiAssist/03_General/div_AiResponseText"),
        Performance_AiAssist_Keywords.AiAssistConstants.FILE_PROMPT
    )

    perfEnd(perfEnabled)
}

			Map ui = [
				iframe        : findTestObject('X360/XAI/Performance/AiAssist/03_General/iframe_Xai'),
				btnAssist     : findTestObject('X360/XAI/Performance/AiAssist/03_General/btn_AiAssist'),

				btnProceed    : findTestObject('X360/XAI/Performance/AiAssist/03_General/btn_UseResponseProceed'),

				response      : findTestObject('X360/XAI/Performance/AiAssist/03_General/div_response'),
				acceptResponse : findTestObject('Object Repository/X360/Misc/Performance/Generic/img_generic', [('sValue') : 'Accept response']),
				feedback: findTestObject('Object Repository/X360/Misc/Performance/Generic/feedback_generic'),
				feedbackOK: findTestObject('Object Repository/X360/XAI/Performance/AiAssist/03_General/btn_feedbackOkay')
			]
		
			try {
				WebUI.switchToDefaultContent()
		
		
				/* Load AI Assist iframe if not yet visible */
				if (!WebUI.waitForElementPresent(ui.iframe, timeout)) {
					GlobalVariable.CellValue = 'Click Ai Assist - Show Card Details'
					CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()
					WebUI.waitForElementClickable(ui.btnAssist, timeout)
					WebUI.click(ui.btnAssist)
					CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
				}
		
		
				/* Switch to AI frame and wait until ready */
				WebUI.switchToFrame(ui.iframe, timeout)
				WebUI.waitForElementNotPresent(ui.busy, 300)
				WebUI.waitForElementPresent(ui.inputBox, timeout)
		
		
				/* Send AI prompt */
				WebUI.setText(ui.inputBox, aiPrompt1.getPromptDetails())
				WebUI.waitForElementClickable(ui.btnSend, timeout)
		
		
				GlobalVariable.CellValue = "Ask a Question - ${aiPrompt1.getPromptDetails()}  Response Received: " + WebUI.getText(ui.response)

				WebUI.click(ui.btnSend)

				CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
		
				/* Validate AI response */
				verifyAIResponse(ui.errorResponse, ui.aiResponse)
		
				/* Send AI prompt */
				WebUI.setText(ui.inputBox, aiPrompt2.getPromptDetails())
				WebUI.waitForElementClickable(ui.btnSend, timeout)
		
				GlobalVariable.CellValue = "Ask a Question - ${aiPrompt2.getPromptDetails()}  Response Received: " + WebUI.getText(ui.aiResponse)
				CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()
				WebUI.click(ui.btnSend)
				WebUI.waitForElementNotPresent(ui.busy, 300)
				CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
		
				/* Validate AI response */
				verifyAIResponse(ui.errorResponse, ui.aiResponse)
		
				downloadConversation()
		
				/*Give Feedback*/
				GlobalVariable.CellValue = "Give Feedback"
				WebUI.waitForElementClickable(ui.acceptResponse, timeout)
				CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()
				WebUI.click(ui.acceptResponse)
				CustomKeywords.'x360.Performance_Functional_Keywords.spinnerWaitToDisappear'(GlobalVariable.sTimeToWait, 2, GlobalVariable.withServerLoad)
				WebUI.waitForElementPresent(ui.feedback, timeout)
				WebUI.setText(ui.feedback, 'Great response!')
				WebUI.waitForElementClickable(ui.feedbackOK, timeout)
				WebUI.click(ui.feedbackOK)
				CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
		
				/* Accept AI response */
				GlobalVariable.CellValue = "Click Use This Response - AI Response Used: " + WebUI.getText(ui.aiResponse)
				WebUI.comment("AI returned response, click Use this response")
				WebUI.waitForElementClickable(ui.useResponse, timeout)
				CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()
				WebUI.click(ui.useResponse)
				WebUI.switchToDefaultContent()
				WebUI.waitForElementClickable(ui.btnProceed, timeout)
				WebUI.click(ui.btnProceed)
				CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
				WebUI.waitForElementNotPresent(ui.iframe, timeout)
		
		
			} catch (Exception e) {
				WebUI.switchToDefaultContent()
				KeywordUtil.markFailed("Unexpected error during AI interaction: ${e.message}")
				e.printStackTrace()
			}
	}

				btnProceed    : findTestObject('X360/XAI/Performance/AiAssist/03_General/btn_UseResponseProceed'),
				errorResponse : findTestObject('X360/XAI/Performance/AiAssist/03_General/div_AiResponseError'),
				aiResponse    : findTestObject('X360/XAI/Performance/AiAssist/03_General/div_AiResponseText'),
				response      : findTestObject('X360/XAI/Performance/AiAssist/03_General/div_response'),
				acceptResponse : findTestObject('Object Repository/X360/Misc/Performance/Generic/accept_Response', [('sValue') : 'Accept response']),
				feedback: findTestObject('Object Repository/X360/Misc/Performance/Generic/feedback_generic'),
				feedbackOK: findTestObject('Object Repository/X360/XAI/Performance/AiAssist/03_General/btn_feedbackOkay')
			]
			
			WebUI.switchToDefaultContent()
		
			try {
				CustomKeywords.'x360.Performance_Functional_Keywords.spinnerWaitToDisappear'(GlobalVariable.sTimeToWait, 2, GlobalVariable.withServerLoad)
				/* Switch to AI frame and wait until ready */
				WebUI.switchToFrame(ui.iframe, timeout)
				CustomKeywords.'x360.Performance_Functional_Keywords.spinnerWaitToDisappear'(GlobalVariable.sTimeToWait, 2, GlobalVariable.withServerLoad)
				WebUI.waitForElementNotPresent(ui.busy, 300)
				WebUI.waitForElementPresent(ui.inputBox, timeout)
		
		
				/* Send AI prompt */
				WebUI.setText(ui.inputBox, aiPrompt1.getPromptDetails())
				WebUI.waitForElementClickable(ui.btnSend, timeout)
				GlobalVariable.CellValue = "Ask a Question - ${aiPrompt1.getPromptDetails()} Response Received: " + WebUI.getText(ui.response)
				CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()
				WebUI.click(ui.btnSend)
				WebUI.waitForElementNotPresent(ui.busy, 300)
				CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
		
				/* Validate AI response */
				verifyAIResponse(ui.errorResponse, ui.aiResponse)
		
				/* Send AI prompt */
				WebUI.setText(ui.inputBox, aiPrompt2.getPromptDetails())
				WebUI.waitForElementClickable(ui.btnSend, timeout)
		
				GlobalVariable.CellValue = "Ask a Question - ${aiPrompt2.getPromptDetails()} Response Received: " + WebUI.getText(ui.aiResponse)
				CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()
				WebUI.click(ui.btnSend)
				WebUI.waitForElementNotPresent(ui.busy, 300)
				CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
		
				/* Validate AI response */
				verifyAIResponse(ui.errorResponse, ui.aiResponse)
		
				downloadConversation()
		
				/*Give Feedback*/
				GlobalVariable.CellValue = "Give Feedback"
				WebUI.scrollToElement(ui.acceptResponse, 5)
				WebUI.mouseOver(ui.acceptResponse)
				WebUI.waitForElementVisible(ui.acceptResponse, timeout)
				WebUI.waitForElementClickable(ui.acceptResponse, timeout)
				CustomKeywords.'x360.Performance_ExcelFunctions.getStartTime'()
				WebUI.click(ui.acceptResponse)
				WebUI.waitForElementPresent(ui.feedback, timeout)
				WebUI.setText(ui.feedback, 'Great response!')
				WebUI.waitForElementClickable(ui.feedbackOK, timeout)
				WebUI.click(ui.feedbackOK)
				CustomKeywords.'x360.Performance_ExcelFunctions.getEndTime'()
		
		
			} catch (Exception e) {
				WebUI.switchToDefaultContent()
				KeywordUtil.markFailed("Unexpected error during AI interaction: ${e.message}")
				e.printStackTrace()
			}
	}
